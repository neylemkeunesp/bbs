\section{Introdução}

\begin{frame}{5.1 Por que Estimação de Parâmetros?}
\begin{block}{Objetivos de Aprendizagem}
\begin{itemize}
    \item Compreender o problema inverso em biologia de sistemas
    \item Dominar métodos de otimização local e global
    \item Aplicar técnicas de estimação a modelos biológicos
    \item Avaliar identificabilidade e incerteza de parâmetros
    \item Validar modelos com critérios estatísticos
\end{itemize}
\end{block}

\vspace{0.3cm}

\definicao{Estimação de Parâmetros}{
Processo de determinar valores numéricos de parâmetros de um modelo matemático a partir de dados experimentais, ajustando o modelo para reproduzir o comportamento observado do sistema.
}
\end{frame}

\begin{frame}{O Problema Inverso}
\begin{columns}
\column{0.5\textwidth}
\textbf{Problema Direto:}
\begin{itemize}
    \item Parâmetros conhecidos
    \item Condições iniciais dadas
    \item Simular $\rightarrow$ Previsões
    \item Determinístico
\end{itemize}

\vspace{0.3cm}

\begin{exampleblock}{Exemplo}
Dados $k_1=0.5$, $k_2=0.1$\\
Simular: $\frac{dx}{dt} = k_1 - k_2 x$\\
$\Rightarrow$ obter $x(t)$
\end{exampleblock}

\column{0.5\textwidth}
\textbf{Problema Inverso:}
\begin{itemize}
    \item Dados experimentais
    \item Parâmetros desconhecidos
    \item Estimar $\leftarrow$ Ajuste
    \item Mal-posto (ill-posed)
\end{itemize}

\vspace{0.3cm}

\begin{alertblock}{Exemplo}
Dados $x(t)$ experimentais\\
Modelo: $\frac{dx}{dt} = k_1 - k_2 x$\\
$\Rightarrow$ encontrar $k_1$, $k_2$
\end{alertblock}
\end{columns}

\vspace{0.3cm}

\importante{
O problema inverso é geralmente mais difícil que o direto!
}
\end{frame}

\begin{frame}{Desafios na Estimação de Parâmetros}
\begin{block}{Principais Obstáculos}
\begin{enumerate}
    \item \textbf{Identificabilidade:} múltiplas soluções possíveis
    \item \textbf{Ruído experimental:} medições imprecisas
    \item \textbf{Dados escassos:} poucas observações
    \item \textbf{Mínimos locais:} otimização não-convexa
    \item \textbf{Correlação de parâmetros:} dificuldade de separação
    \item \textbf{Custo computacional:} simulações repetidas
    \item \textbf{Incerteza:} quantificação de confiança
\end{enumerate}
\end{block}

\section{Formulação do Problema}

\begin{frame}{Função Objetivo}
\definicao{Função Objetivo (Cost Function)}{
Medida quantitativa da discrepância entre dados experimentais e previsões do modelo. Objetivo: minimizar esta função.
}

\vspace{0.3cm}

\begin{block}{Componentes}
\begin{itemize}
    \item $y_i$: dados experimentais (observações)
    \item $\hat{y}_i(\vect{\theta})$: previsões do modelo
    \item $\vect{\theta}$: vetor de parâmetros a estimar
    \item $n$: número de observações
\end{itemize}
\end{block}

\destaque{
\begin{equation*}
J(\vect{\theta}) = \text{medida de erro entre } \{y_i\} \text{ e } \{\hat{y}_i(\vect{\theta})\}
\end{equation*}
}

\textbf{Problema de Otimização:}
\begin{equation*}
\vect{\theta}^* = \arg\min_{\vect{\theta}} J(\vect{\theta})
\end{equation*}
\end{frame}

\begin{frame}{Mínimos Quadrados Ordinários (OLS)}
\begin{block}{Método Mais Comum}
Minimizar a soma dos quadrados dos resíduos:
\end{block}

\destaque{
\begin{equation*}
J(\vect{\theta}) = \text{SSE}(\vect{\theta}) = \sum_{i=1}^{n} [y_i - \hat{y}_i(\vect{\theta})]^2
\end{equation*}
}

\begin{columns}
\column{0.5\textwidth}
\textbf{Vantagens:}
\begin{itemize}
    \item Simples e intuitivo
    \item Bem estabelecido
    \item Solução analítica (modelos lineares)
    \item Interpretação estatística
\end{itemize}

\column{0.5\textwidth}
\textbf{Suposições:}
\begin{itemize}
    \item Erros gaussianos
    \item Variância constante
    \item Erros independentes
    \item Outliers problemáticos
\end{itemize}
\end{columns}

\vspace{0.3cm}

\begin{exampleblock}{Resíduo}
$r_i = y_i - \hat{y}_i(\vect{\theta})$ = erro de previsão no ponto $i$
\end{exampleblock}
\end{frame}

\begin{frame}{Mínimos Quadrados Ponderados (WLS)}
\begin{block}{Quando usar?}
Quando as observações têm diferentes níveis de incerteza ou confiabilidade
\end{block}

\destaque{
\begin{equation*}
J(\vect{\theta}) = \sum_{i=1}^{n} w_i [y_i - \hat{y}_i(\vect{\theta})]^2
\end{equation*}
}

\textbf{Pesos $w_i$:}
\begin{itemize}
    \item $w_i = 1/\sigma_i^2$: inverso da variância (comum)
    \item $w_i = 1/y_i$: proporcional ao valor (dados log-scale)
    \item $w_i$ customizado: baseado em conhecimento experimental
\end{itemize}

\vspace{0.3cm}

\exemplo{Cinética Enzimática}{
Concentrações baixas têm maior erro relativo\\
$\Rightarrow$ usar pesos $w_i = 1/\hat{y}_i$ ou $w_i = 1/\hat{y}_i^2$
}
\end{frame}

\begin{frame}{Maximum Likelihood Estimation (MLE)}
\begin{block}{Abordagem Estatística}
Encontrar parâmetros que maximizam a probabilidade (likelihood) de observar os dados
\end{block}

\textbf{Função de Verossimilhança:}
\begin{equation*}
\mathcal{L}(\vect{\theta}) = P(\text{dados} | \vect{\theta})
\end{equation*}

\textbf{Log-verossimilhança (mais conveniente):}
\begin{equation*}
\ln \mathcal{L}(\vect{\theta}) = \sum_{i=1}^{n} \ln P(y_i | \vect{\theta})
\end{equation*}

\vspace{0.3cm}

\begin{exampleblock}{Caso Gaussiano}
Se erros $\sim \mathcal{N}(0, \sigma^2)$:
\begin{equation*}
\ln \mathcal{L}(\vect{\theta}) = -\frac{n}{2}\ln(2\pi\sigma^2) - \frac{1}{2\sigma^2}\sum_{i=1}^{n}[y_i - \hat{y}_i(\vect{\theta})]^2
\end{equation*}
Maximizar $\ln \mathcal{L}$ = Minimizar SSE!
\end{exampleblock}
\end{frame}

\begin{frame}{Abordagem Bayesiana}
\begin{block}{Incorpora Conhecimento Prévio}
Usa teorema de Bayes para atualizar crenças sobre parâmetros
\end{block}

\destaque{
\begin{equation*}
P(\vect{\theta} | \text{dados}) = \frac{P(\text{dados} | \vect{\theta}) \cdot P(\vect{\theta})}{P(\text{dados})}
\end{equation*}
}

\begin{itemize}
    \item $P(\vect{\theta} | \text{dados})$: \textbf{posterior} (o que queremos)
    \item $P(\text{dados} | \vect{\theta})$: \textbf{likelihood} (ajuste aos dados)
    \item $P(\vect{\theta})$: \textbf{prior} (conhecimento prévio)
    \item $P(\text{dados})$: constante de normalização
\end{itemize}

\vspace{0.3cm}

\begin{exampleblock}{Vantagens}
\begin{itemize}
    \item Incorpora informação da literatura
    \item Quantifica incerteza naturalmente
    \item Previne overfitting (regularização)
\end{itemize}
\end{exampleblock}

\begin{alertblock}{Desafio}
Requer escolha de distribuições priors e métodos computacionais intensivos (MCMC)
\end{alertblock}
\end{frame}

\section{Busca em Grade (Grid Search)}

\begin{frame}{Conceito de Busca em Grade}
\definicao{Grid Search}{
Método de otimização que avalia sistematicamente a função objetivo em uma grade regular de valores de parâmetros.
}

\begin{center}
\begin{tikzpicture}[scale=0.9]
    % 2D grid
    \draw[step=0.8cm, gray, very thin] (0,0) grid (4,4);
    \draw[->] (0,0) -- (4.5,0) node[right] {$\theta_1$};
    \draw[->] (0,0) -- (0,4.5) node[above] {$\theta_2$};

    % Grid points with values
    \foreach \x in {0.4, 1.2, 2.0, 2.8, 3.6} {
        \foreach \y in {0.4, 1.2, 2.0, 2.8, 3.6} {
            \pgfmathsetmacro{\val}{100*exp(-(\x-2)^2/2 - (\y-2.4)^2/2)}
            \pgfmathparse{\val>50 ? 1 : 0}
            \ifnum\pgfmathresult=1
                \fill[verdeacido] (\x,\y) circle (2pt);
            \else
                \pgfmathparse{\val>20 ? 1 : 0}
                \ifnum\pgfmathresult=1
                    \fill[laranjacel] (\x,\y) circle (2pt);
                \else
                    \fill[red] (\x,\y) circle (2pt);
                \fi
            \fi
        }
    }

    % Best point
    \fill[blue] (2,2.4) circle (3pt) node[right, font=\tiny] {Mínimo};

    \node[below, font=\tiny] at (2,-0.5) {Grade 5x5 = 25 avaliações};
\end{tikzpicture}
\end{center}

\textbf{Algoritmo:}
\begin{enumerate}
    \item Definir intervalos para cada parâmetro
    \item Criar grade com $n_1 \times n_2 \times \ldots \times n_p$ pontos
    \item Avaliar $J(\vect{\theta})$ em cada ponto
    \item Selecionar ponto com menor $J$
\end{enumerate}
\end{frame}

\begin{frame}{Vantagens e Desvantagens}
\begin{columns}
\column{0.5\textwidth}
\begin{block}{Vantagens}
\begin{itemize}
    \item Simples de implementar
    \item Facilmente paralelizável
    \item Garantia de exploração
    \item Visualização intuitiva
    \item Sem derivadas necessárias
\end{itemize}
\end{block}

\column{0.5\textwidth}
\begin{alertblock}{Desvantagens}
\begin{itemize}
    \item Custo exponencial: $O(n^p)$
    \item Inviável para muitos parâmetros
    \item Resolução limitada
    \item Desperdício de avaliações
    \item Maldição da dimensionalidade
\end{itemize}
\end{alertblock}
\end{columns}

\vspace{0.3cm}

\importante{
\textbf{Custo Computacional:}\\
Com 10 valores por parâmetro:
\begin{itemize}
    \item 2 parâmetros: $10^2 = 100$ avaliações
    \item 5 parâmetros: $10^5 = 100.000$ avaliações
    \item 10 parâmetros: $10^{10}$ avaliações (impraticável!)
\end{itemize}
}
\end{frame}

\begin{frame}[fragile]{Implementação: Grid Search 1D}
\begin{lstlisting}[language=Python, caption=Busca em grade unidimensional, basicstyle=\ttfamily\footnotesize]
import numpy as np
import matplotlib.pyplot as plt

def objective_function(theta):
    """Funcao objetivo a minimizar"""
    return (theta - 3)**2 + 5

# Definir grade
theta_values = np.linspace(0, 6, 50)

# Avaliar funcao objetivo
J_values = [objective_function(theta) for theta in theta_values]

# Encontrar minimo
idx_min = np.argmin(J_values)
theta_best = theta_values[idx_min]
J_best = J_values[idx_min]

print(f"Parametro otimo: {theta_best:.3f}")
print(f"Valor da funcao: {J_best:.3f}")

# Visualizar
plt.figure(figsize=(10, 6))
plt.plot(theta_values, J_values, 'b-', linewidth=2)
plt.plot(theta_best, J_best, 'ro', markersize=10,
         label=f'Minimo: theta={theta_best:.2f}')
plt.xlabel('Parametro theta')
plt.ylabel('J(theta)')
plt.legend()
plt.grid(True)
\end{lstlisting}
\end{frame}

def sse_model(params, t_data, y_data):
    """SSE para modelo exponencial y = a*exp(-b*t)"""
    a, b = params
    y_pred = a * np.exp(-b * t_data)
    sse = np.sum((y_data - y_pred)**2)
    return sse

# Dados experimentais
t_data = np.array([0, 1, 2, 3, 4, 5])
y_data = np.array([10, 6.1, 3.7, 2.2, 1.4, 0.8])

# Grade de parametros
a_range = np.linspace(5, 15, 20)
b_range = np.linspace(0.1, 1.0, 20)

# Avaliar em toda a grade
SSE_grid = np.zeros((len(a_range), len(b_range)))

for i, a in enumerate(a_range):
    for j, b in enumerate(b_range):
        SSE_grid[i, j] = sse_model([a, b], t_data, y_data)

# Encontrar minimo
min_idx = np.unravel_index(SSE_grid.argmin(), SSE_grid.shape)
a_best = a_range[min_idx[0]]
b_best = b_range[min_idx[1]]

print(f"Parametros otimos: a={a_best:.3f}, b={b_best:.3f}")
\end{lstlisting}
\end{frame}

\section{Métodos de Gradiente}

\begin{frame}{Gradient Descent (Descida do Gradiente)}
\definicao{Gradient Descent}{
Método iterativo que se move na direção do negativo do gradiente para encontrar um mínimo local da função objetivo.
}

\destaque{
\begin{equation*}
\vect{\theta}^{(k+1)} = \vect{\theta}^{(k)} - \alpha \nabla J(\vect{\theta}^{(k)})
\end{equation*}
}

\begin{itemize}
    \item $\vect{\theta}^{(k)}$: parâmetros na iteração $k$
    \item $\alpha$: taxa de aprendizado (step size)
    \item $\nabla J = \left[\frac{\partial J}{\partial \theta_1}, \frac{\partial J}{\partial \theta_2}, \ldots\right]^T$: gradiente
\end{itemize}

\vspace{0.3cm}

\end{frame}

\importante{

}

\vspace{0.3cm}

\begin{block}{Estratégias}
\begin{itemize}
    \item \textbf{Fixo:} $\alpha$ constante (simples, pode não convergir)
    \item \textbf{Decrescente:} $\alpha_k = \alpha_0 / (1+k)$ (garante convergência)
    \item \textbf{Line search:} otimizar $\alpha$ em cada iteração (mais robusto)
    \item \textbf{Adaptativo:} ajustar baseado em progresso (Adam, RMSprop)
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Critérios de Convergência}
\begin{block}{Quando Parar?}
Teste múltiplos critérios simultaneamente:
\end{block}

\begin{enumerate}
    \item \textbf{Gradiente pequeno:} $\|\nabla J(\vect{\theta}^{(k)})\| < \epsilon_g$
    \begin{itemize}
        \item Indica ponto estacionário
    \end{itemize}

    \item \textbf{Mudança em parâmetros:} $\|\vect{\theta}^{(k+1)} - \vect{\theta}^{(k)}\| < \epsilon_\theta$
    \begin{itemize}
        \item Parâmetros não mudam mais
    \end{itemize}

    \item \textbf{Mudança na função:} $|J^{(k+1)} - J^{(k)}| < \epsilon_J$
    \begin{itemize}
        \item Função objetivo estabilizou
    \end{itemize}

    \item \textbf{Número máximo de iterações:} $k > k_{\max}$
    \begin{itemize}
        \item Prevenir loops infinitos
    \end{itemize}
\end{enumerate}

\begin{exampleblock}{Valores Típicos}
$\epsilon_g = 10^{-6}$, $\epsilon_\theta = 10^{-6}$, $\epsilon_J = 10^{-8}$, $k_{\max} = 1000$
\end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Implementação: Gradient Descent}
\begin{lstlisting}[language=Python, caption=Descida do gradiente com derivadas numéricas, basicstyle=\ttfamily\footnotesize]
def gradient_descent(f, theta0, alpha=0.01, max_iter=1000,
                      tol=1e-6):
    """
    Gradient descent com derivadas numericas
    f: funcao objetivo
    theta0: chute inicial
    """
    theta = np.array(theta0, dtype=float)
    history = [theta.copy()]

    for k in range(max_iter):
        # Calcular gradiente numericamente
        grad = numerical_gradient(f, theta)

        # Atualizar parametros
        theta_new = theta - alpha * grad

        # Checar convergencia
        if np.linalg.norm(theta_new - theta) < tol:
            print(f"Convergiu em {k} iteracoes")
            break

        theta = theta_new
        history.append(theta.copy())

    return theta, np.array(history)

def numerical_gradient(f, theta, h=1e-5):
    """Gradiente por diferencas finitas"""
    grad = np.zeros_like(theta)
    for i in range(len(theta)):
        theta_plus = theta.copy()
        theta_plus[i] += h
        grad[i] = (f(theta_plus) - f(theta)) / h
    return grad
\end{lstlisting}
\end{frame}

\begin{frame}{Método de Newton-Raphson}
\begin{block}{Ideia}
Usar informação de segunda ordem (curvatura) para convergência mais rápida
\end{block}

\destaque{
\begin{equation*}
\vect{\theta}^{(k+1)} = \vect{\theta}^{(k)} - \mathbf{H}^{-1}(\vect{\theta}^{(k)}) \nabla J(\vect{\theta}^{(k)})
\end{equation*}
}

\begin{itemize}
    \item $\mathbf{H}$: matriz Hessiana (derivadas segundas)
    \item $H_{ij} = \frac{\partial^2 J}{\partial \theta_i \partial \theta_j}$
\end{itemize}

\vspace{0.3cm}

\begin{columns}
\column{0.5\textwidth}
\textbf{Vantagens:}
\begin{itemize}
    \item Convergência quadrática
    \item Muito rápido perto do mínimo
    \item Não precisa de step size
\end{itemize}

\column{0.5\textwidth}
\textbf{Desvantagens:}
\begin{itemize}
    \item Calcular $\mathbf{H}$ é caro
    \item Inverter $\mathbf{H}$ também
    \item Pode não convergir longe do mínimo
    \item Requer $\mathbf{H}$ positiva definida
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Método de Gauss-Newton}
\begin{block}{Específico para Mínimos Quadrados}
Aproxima Hessiana usando Jacobiano das previsões do modelo
\end{block}

Para $J(\vect{\theta}) = \sum_{i=1}^{n} r_i^2(\vect{\theta})$ onde $r_i = y_i - \hat{y}_i(\vect{\theta})$:

\destaque{
\begin{equation*}
\vect{\theta}^{(k+1)} = \vect{\theta}^{(k)} + (\mathbf{J}^T\mathbf{J})^{-1}\mathbf{J}^T \vect{r}
\end{equation*}
}

\begin{itemize}
    \item $\mathbf{J}$: matriz Jacobiana dos resíduos ($J_{ij} = \partial r_i / \partial \theta_j$)
    \item $\vect{r}$: vetor de resíduos
\end{itemize}

\vspace{0.3cm}

\begin{exampleblock}{Vantagens}
\begin{itemize}
    \item Evita calcular Hessiana completa
    \item Boa performance para problemas de mínimos quadrados
    \item Usado como base do Levenberg-Marquardt
\end{itemize}
\end{exampleblock}
\end{frame}

\begin{frame}{Levenberg-Marquardt (LM)}
\definicao{Levenberg-Marquardt}{
Combina Gauss-Newton e gradient descent, interpolando entre ambos via parâmetro de amortecimento.
}

\destaque{
\begin{equation*}
\vect{\theta}^{(k+1)} = \vect{\theta}^{(k)} + (\mathbf{J}^T\mathbf{J} + \lambda \mathbf{I})^{-1}\mathbf{J}^T \vect{r}
\end{equation*}
}

\begin{itemize}
    \item $\lambda$: parâmetro de amortecimento (damping)
    \item $\lambda$ grande $\Rightarrow$ gradient descent (pequenos passos, robusto)
    \item $\lambda$ pequeno $\Rightarrow$ Gauss-Newton (passos grandes, rápido)
\end{itemize}

\vspace{0.3cm}

\textbf{Estratégia Adaptativa:}
\begin{itemize}
    \item Se $J$ diminui: aceitar passo, diminuir $\lambda$ (mais agressivo)
    \item Se $J$ aumenta: rejeitar passo, aumentar $\lambda$ (mais conservador)
\end{itemize}

\importante{
LM é o método padrão para ajuste não-linear de curvas!
}
\end{frame}

\begin{frame}[fragile]{Exemplo: Michaelis-Menten com scipy}
\begin{lstlisting}[language=Python, caption=Ajuste de parâmetros com Levenberg-Marquardt, basicstyle=\ttfamily\footnotesize]
from scipy.optimize import least_squares, curve_fit
import numpy as np

# Modelo de Michaelis-Menten
def michaelis_menten(S, Vmax, Km):
    return Vmax * S / (Km + S)

# Dados experimentais
S_data = np.array([0.5, 1, 2, 5, 10, 20, 50])
v_data = np.array([0.4, 0.7, 1.2, 1.8, 2.1, 2.3, 2.4])

# Metodo 1: curve_fit (interface simples)
params, cov = curve_fit(michaelis_menten, S_data, v_data,
                         p0=[3, 5])
Vmax_fit, Km_fit = params

print(f"Vmax = {Vmax_fit:.3f}")
print(f"Km = {Km_fit:.3f}")

# Metodo 2: least_squares (mais controle)
def residuals(params):
    Vmax, Km = params
    return v_data - michaelis_menten(S_data, Vmax, Km)

result = least_squares(residuals, x0=[3, 5], method='lm')
print(f"Parametros otimos: {result.x}")
print(f"Custo final: {result.cost:.6f}")
print(f"Sucesso: {result.success}")
\end{lstlisting}
\end{frame}

\textbf{Quando Usar Otimização Global?}
\begin{itemize}
    \item Função objetivo altamente não-linear
    \item Múltiplos mínimos locais
    \item Sem boa estimativa inicial
    \item Parâmetros com grande incerteza
    \item Modelos biológicos complexos
\end{itemize}
\end{frame}

\begin{frame}{Algoritmos Genéticos (GA)}
\definicao{Algoritmo Genético}{
Método inspirado na evolução biológica que mantém uma população de soluções candidatas e as evolui usando seleção, cruzamento e mutação.
}

\textbf{Analogia Biológica:}
\begin{itemize}
    \item \textbf{Indivíduo:} conjunto de parâmetros $\vect{\theta}$
    \item \textbf{Gene:} parâmetro individual $\theta_i$
    \item \textbf{Fitness:} inverso da função objetivo $1/J(\vect{\theta})$
    \item \textbf{População:} conjunto de soluções candidatas
\end{itemize}

\vspace{0.3cm}

\begin{block}{Operadores Genéticos}
\begin{enumerate}
    \item \textbf{Seleção:} escolher indivíduos com melhor fitness
    \item \textbf{Cruzamento:} combinar dois pais para gerar filhos
    \item \textbf{Mutação:} introduzir variação aleatória
    \item \textbf{Elitismo:} preservar melhores soluções
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{Simulated Annealing (SA)}
\begin{block}{Inspiração}
Processo de recozimento em metalurgia: aquecer metal e esfriar lentamente para alcançar estrutura cristalina de mínima energia
\end{block}

\textbf{Ideia Principal:}
\begin{itemize}
    \item Permitir movimentos para soluções piores com probabilidade decrescente
    \item "Temperatura" controla aceitação de soluções piores
    \item Alta temperatura: exploração ampla
    \item Baixa temperatura: refinamento local
\end{itemize}

\destaque{
\begin{equation*}
P(\text{aceitar}) = \begin{cases}
1 & \text{se } \Delta J < 0 \\
e^{-\Delta J / T} & \text{se } \Delta J \geq 0
\end{cases}
\end{equation*}
}

onde $\Delta J = J(\vect{\theta}_{\text{novo}}) - J(\vect{\theta}_{\text{atual}})$ e $T$ é a temperatura.

\begin{exampleblock}{Cronograma de Resfriamento}
$T_k = T_0 \cdot \alpha^k$ onde $0 < \alpha < 1$ (ex: $\alpha = 0.95$)
\end{exampleblock}
\end{frame}

\begin{frame}{Differential Evolution (DE)}
\definicao{Evolução Diferencial}{
Algoritmo evolutivo que cria novos candidatos através de diferenças entre membros da população.
}

\textbf{Operação Principal:}
\begin{enumerate}
    \item Selecionar 3 indivíduos aleatórios: $\vect{a}$, $\vect{b}$, $\vect{c}$
    \item Criar vetor de mutação: $\vect{v} = \vect{a} + F(\vect{b} - \vect{c})$
    \item Cruzamento: $\vect{u}_i = \begin{cases} v_i & \text{se } rand < CR \\ \theta_i & \text{caso contrário} \end{cases}$
    \item Seleção: manter melhor entre atual e novo
\end{enumerate}

\begin{itemize}
    \item $F$: fator de escala (diferential weight), tipicamente $F \in [0.5, 1]$
    \item $CR$: taxa de cruzamento (crossover rate), tipicamente $CR \in [0.5, 0.9]$
\end{itemize}

\vspace{0.3cm}

\begin{exampleblock}{Vantagens}
\begin{itemize}
    \item Simples de implementar
    \item Poucos parâmetros para ajustar
    \item Boa performance em muitos problemas
    \item Robusto e confiável
\end{itemize}
\end{exampleblock}
\end{frame}

\begin{frame}{Particle Swarm Optimization (PSO)}
\begin{block}{Inspiração}
Comportamento social de bandos de pássaros ou cardumes de peixes
\end{block}

\textbf{Conceito:}
\begin{itemize}
    \item Cada partícula tem posição $\vect{x}_i$ e velocidade $\vect{v}_i$
    \item Partículas se movem influenciadas por:
    \begin{itemize}
        \item Sua melhor posição histórica ($\vect{p}_i$)
        \item Melhor posição global do enxame ($\vect{g}$)
    \end{itemize}
\end{itemize}

\destaque{
\begin{align*}
\vect{v}_i^{(k+1)} &= w\vect{v}_i^{(k)} + c_1 r_1 (\vect{p}_i - \vect{x}_i^{(k)}) + c_2 r_2 (\vect{g} - \vect{x}_i^{(k)})\\
\vect{x}_i^{(k+1)} &= \vect{x}_i^{(k)} + \vect{v}_i^{(k+1)}
\end{align*}
}

\begin{itemize}
    \item $w$: peso de inércia
    \item $c_1$, $c_2$: coeficientes cognitivo e social
    \item $r_1$, $r_2$: números aleatórios em $[0,1]$
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Implementação: Differential Evolution}
\begin{lstlisting}[language=Python, caption=DE usando scipy.optimize, basicstyle=\ttfamily\footnotesize]
from scipy.optimize import differential_evolution

# Funcao objetivo
def objective(params, t_data, y_data):
    """Modelo exponencial: y = a * exp(-b*t)"""
    a, b = params
    y_pred = a * np.exp(-b * t_data)
    sse = np.sum((y_data - y_pred)**2)
    return sse

# Dados experimentais
t_data = np.array([0, 1, 2, 3, 4, 5])
y_data = np.array([10, 6.1, 3.7, 2.2, 1.4, 0.8])

# Definir bounds para parametros
bounds = [(1, 20), (0.01, 2.0)]  # [(a_min, a_max), (b_min, b_max)]

# Otimizar com DE
result = differential_evolution(
    objective,
    bounds,
    args=(t_data, y_data),
    strategy='best1bin',
    maxiter=1000,
    popsize=15,
    tol=1e-7,
    mutation=(0.5, 1),
    recombination=0.7,
    seed=42
)

print(f"Parametros otimos: a={result.x[0]:.3f}, b={result.x[1]:.3f}")
print(f"SSE minimo: {result.fun:.6f}")
print(f"Convergiu: {result.success}")
\end{lstlisting}
\end{frame}

\section{Análise de Identificabilidade}

\begin{frame}{Identificabilidade Estrutural}
\definicao{Identificabilidade Estrutural}{
Propriedade teórica do modelo: os parâmetros podem ser determinados unicamente a partir de dados perfeitos (infinitos e sem ruído)?
}

\vspace{0.3cm}

\begin{columns}
\column{0.5\textwidth}
\begin{block}{Identificável}
\begin{equation*}
\frac{dx}{dt} = k_1 - k_2 x
\end{equation*}
Ambos $k_1$ e $k_2$ podem ser determinados unicamente
\end{block}

\column{0.5\textwidth}
\begin{alertblock}{Não Identificável}
\begin{equation*}
\frac{dx}{dt} = (k_1 k_2) - k_2 x
\end{equation*}
Apenas o produto $k_1 k_2$ é identificável, não $k_1$ e $k_2$ separadamente
\end{alertblock}
\end{columns}

\vspace{0.3cm}

\importante{
Antes de estimar parâmetros, verifique se o modelo é estruturalmente identificável!
}

\textbf{Ferramentas:}
\begin{itemize}
    \item Análise algébrica (eliminação de variáveis)
    \item Software especializado: DAISY, GenSSI
\end{itemize}
\end{frame}

\begin{frame}{Identificabilidade Prática}
\definicao{Identificabilidade Prática}{
Podem os parâmetros ser determinados com precisão razoável a partir de dados experimentais reais (finitos e com ruído)?
}

\vspace{0.3cm}

\textbf{Causas de Não-Identificabilidade Prática:}
\begin{itemize}
    \item Dados insuficientes ou de baixa qualidade
    \item Parâmetros altamente correlacionados
    \item Modelo super-parametrizado (overfitting)
    \item Sensibilidade muito baixa a alguns parâmetros
\end{itemize}

\vspace{0.3cm}

\exemplo{Correlação de Parâmetros}{
Modelo: $y = a \cdot e^{-b \cdot t}$\\
Se $t$ é pequeno: $e^{-bt} \approx 1 - bt$\\
$\Rightarrow y \approx a(1 - bt) = a - abt$\\
Difícil separar $a$ de $ab$ (produto correlacionado)
}
\end{frame}

\begin{frame}{Matriz de Informação de Fisher (FIM)}
\begin{block}{Quantifica Informação sobre Parâmetros}
Mede quanta informação os dados contêm sobre cada parâmetro
\end{block}

\destaque{
\begin{equation*}
\mathbf{F}_{ij} = \sum_{k=1}^{n} \frac{1}{\sigma_k^2} \frac{\partial \hat{y}_k}{\partial \theta_i} \frac{\partial \hat{y}_k}{\partial \theta_j}
\end{equation*}
}

onde:
\begin{itemize}
    \item $\hat{y}_k$: previsão do modelo no ponto $k$
    \item $\sigma_k^2$: variância do erro no ponto $k$
    \item $\theta_i$, $\theta_j$: parâmetros
\end{itemize}

\vspace{0.3cm}

\textbf{Interpretação:}
\begin{itemize}
    \item FIM grande $\Rightarrow$ alta informação, parâmetros bem determinados
    \item FIM pequena $\Rightarrow$ baixa informação, alta incerteza
    \item $\text{det}(\mathbf{F}) \approx 0$ $\Rightarrow$ parâmetros não identificáveis
\end{itemize}
\end{frame}

\begin{frame}{Intervalos de Confiança}
\begin{block}{Estimativa Assintótica}
Para grandes amostras, $\vect{\theta} \sim \mathcal{N}(\vect{\theta}^*, \mathbf{C})$ onde:
\end{block}

\destaque{
\begin{equation*}
\mathbf{C} = \text{Cov}(\vect{\theta}) \approx \mathbf{F}^{-1}
\end{equation*}
}

\textbf{Intervalo de Confiança (95\%):}
\begin{equation*}
\theta_i^* \pm 1.96 \sqrt{C_{ii}}
\end{equation*}

\vspace{0.3cm}

\begin{exampleblock}{Na Prática}
\texttt{scipy.optimize.curve\_fit} retorna matriz de covariância:
\begin{itemize}
    \item Diagonal: $\sqrt{C_{ii}}$ = erro padrão do parâmetro $i$
    \item Off-diagonal: $C_{ij}$ = correlação entre parâmetros $i$ e $j$
\end{itemize}
\end{exampleblock}

\importante{
Intervalos de confiança estreitos $\not\Rightarrow$ modelo correto!\\
Apenas indicam precisão do ajuste.
}
\end{frame}

\begin{frame}{Profile Likelihood}
\definicao{Profile Likelihood}{
Método robusto para calcular intervalos de confiança explorando o espaço de parâmetros sistematicamente.
}

\textbf{Procedimento para parâmetro $\theta_i$:}
\begin{enumerate}
    \item Fixar $\theta_i$ em valor específico
    \item Otimizar todos os outros parâmetros
    \item Registrar $J_{\min}(\theta_i)$
    \item Repetir para vários valores de $\theta_i$
    \item Plotar $J_{\min}$ vs $\theta_i$
\end{enumerate}

\vspace{0.3cm}

\begin{center}
\begin{tikzpicture}[scale=0.8]
    \draw[->] (0,0) -- (6,0) node[right] {$\theta_i$};
    \draw[->] (0,0) -- (0,3) node[above] {$J_{\min}$};

    % Profile likelihood
    \draw[thick, azulescuro, domain=0.5:5.5, samples=100, smooth]
        plot (\x, {0.5*(\x-3)^2 + 0.3});

    % Threshold for 95% CI
    \draw[dashed, red] (0,1.23) -- (6,1.23) node[right, font=\tiny] {$\chi^2$ threshold};

    % Confidence interval
    \draw[<->, thick, verdeacido] (1.5,0.1) -- (4.5,0.1);
    \node[below, font=\tiny] at (3,0.1) {IC 95\%};

    \fill[red] (3,0.3) circle (2pt) node[below, font=\tiny] {$\theta_i^*$};
\end{tikzpicture}
\end{center}
\end{frame}

\begin{frame}[fragile]{Análise de Correlação de Parâmetros}
\begin{lstlisting}[language=Python, caption=Matriz de correlação, basicstyle=\ttfamily\footnotesize]

\end{lstlisting}
\end{frame}

\section{Validação e Critérios de Ajuste}

\begin{frame}{Análise de Resíduos}
\begin{block}{Verificação de Hipóteses}
Examinar resíduos $r_i = y_i - \hat{y}_i$ para validar suposições do modelo
\end{block}

\textbf{Padrões a Verificar:}
\begin{columns}
\column{0.5\textwidth}
\begin{exampleblock}{Bom Ajuste}
\begin{itemize}
    \item Distribuição aleatória
    \item Sem padrão sistemático
    \item Centrados em zero
    \item Variância constante
    \item Aproximadamente normais
\end{itemize}
\end{exampleblock}

\column{0.5\textwidth}
\begin{alertblock}{Problemas}
\begin{itemize}
    \item Padrão curvo: modelo inadequado
    \item Funil: heterocedasticidade
    \item Outliers: dados anômalos
    \item Não-normalidade: transformar dados
\end{itemize}
\end{alertblock}
\end{columns}

\begin{frame}{Coeficiente de Determinação ($R^2$)}
\definicao{$R^2$}{
Proporção da variância nos dados explicada pelo modelo. Varia de 0 (nenhum ajuste) a 1 (ajuste perfeito).
}

\destaque{
\begin{equation*}
R^2 = 1 - \frac{\text{SS}_{\text{res}}}{\text{SS}_{\text{tot}}} = 1 - \frac{\sum_i (y_i - \hat{y}_i)^2}{\sum_i (y_i - \bar{y})^2}
\end{equation*}
}

\begin{itemize}
    \item $\text{SS}_{\text{res}}$: soma dos quadrados dos resíduos
    \item $\text{SS}_{\text{tot}}$: soma dos quadrados totais
    \item $\bar{y}$: média dos dados
\end{itemize}

\vspace{0.3cm}

\begin{alertblock}{ATENÇÃO!}
\begin{itemize}
    \item $R^2$ sempre aumenta ao adicionar parâmetros (mesmo irrelevantes)
    \item $R^2$ alto não garante modelo correto ou preditivo
    \item Pode ser enganoso para modelos não-lineares
    \item Use $R^2$ ajustado para comparar modelos com diferentes números de parâmetros
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{$R^2$ Ajustado}
\begin{block}{Penaliza Complexidade do Modelo}
\end{block}

\destaque{
\begin{equation*}
R^2_{\text{adj}} = 1 - \frac{(1-R^2)(n-1)}{n-p-1}
\end{equation*}
}

onde:
\begin{itemize}
    \item $n$: número de observações
    \item $p$: número de parâmetros
\end{itemize}

\vspace{0.3cm}

\textbf{Características:}
\begin{itemize}
    \item Pode diminuir ao adicionar parâmetros ruins
    \item Mais apropriado para comparação de modelos
    \item Ainda limitado como métrica única
\end{itemize}

\begin{exampleblock}{Exemplo}
Modelo A: $R^2 = 0.95$, $p = 10$\\
Modelo B: $R^2 = 0.93$, $p = 3$\\
Para $n = 50$: $R^2_{\text{adj,A}} = 0.936$, $R^2_{\text{adj,B}} = 0.925$\\
$\Rightarrow$ Modelo B pode ser preferível (mais parcimonioso)
\end{exampleblock}
\end{frame}

\begin{frame}{Akaike Information Criterion (AIC)}
\definicao{AIC}{
Critério que balanceia qualidade do ajuste com complexidade do modelo, baseado em teoria da informação.
}

\destaque{
\begin{equation*}
\text{AIC} = 2p - 2\ln(\mathcal{L}) \approx n\ln\left(\frac{\text{SSE}}{n}\right) + 2p
\end{equation*}
}

\begin{itemize}
    \item $p$: número de parâmetros
    \item $\mathcal{L}$: máxima verossimilhança
    \item SSE: soma dos quadrados dos erros
\end{itemize}

\vspace{0.3cm}

\textbf{Interpretação:}
\begin{itemize}
    \item Menor AIC = melhor modelo
    \item Penaliza adição de parâmetros
    \item Diferenças: $\Delta\text{AIC} > 10$ = evidência forte
\end{itemize}

\begin{exampleblock}{AIC Corrigido (AICc)}
Para amostras pequenas ($n/p < 40$):
\begin{equation*}
\text{AICc} = \text{AIC} + \frac{2p(p+1)}{n-p-1}
\end{equation*}
\end{exampleblock}
\end{frame}

\begin{frame}{Bayesian Information Criterion (BIC)}
\definicao{BIC (ou Schwarz Criterion)}{
Similar ao AIC, mas penaliza mais fortemente a complexidade do modelo.
}

\destaque{
\begin{equation*}
\text{BIC} = p\ln(n) - 2\ln(\mathcal{L}) \approx n\ln\left(\frac{\text{SSE}}{n}\right) + p\ln(n)
\end{equation*}
}

\vspace{0.3cm}

\textbf{Comparação AIC vs BIC:}
\begin{itemize}
    \item BIC penaliza mais complexidade que AIC (termo $\ln(n)$ vs $2$)
    \item BIC tende a selecionar modelos mais simples
    \item AIC minimiza erro de previsão; BIC busca "modelo verdadeiro"
    \item Para $n > 7$: BIC penaliza mais que AIC
\end{itemize}

\vspace{0.3cm}

\begin{center}
\begin{tabular}{lcc}
\toprule
\textbf{Critério} & \textbf{Objetivo} & \textbf{Preferência} \\
\midrule
AIC & Previsão & Modelos maiores \\
BIC & Seleção & Modelos menores \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}{Cross-Validation (Validação Cruzada)}
\begin{block}{Teste de Capacidade Preditiva}
Avaliar desempenho do modelo em dados não vistos durante o treinamento
\end{block}

\textbf{K-Fold Cross-Validation:}
\begin{enumerate}
    \item Dividir dados em $K$ subconjuntos (folds)
    \item Para cada fold $k$:
    \begin{itemize}
        \item Treinar modelo nos outros $K-1$ folds
        \item Testar no fold $k$
        \item Calcular erro de previsão
    \end{itemize}
    \item Calcular erro médio de CV
\end{enumerate}

\vspace{0.3cm}

}

\begin{alertblock}{Sintomas de Overfitting}
\begin{itemize}
    \item Erro de treinamento muito menor que erro de teste
    \item Modelo complexo com muitos parâmetros
    \item Intervalos de confiança muito largos
    \item Previsões ruins fora do range dos dados
\end{itemize}
\end{alertblock}

\begin{frame}[fragile]{Implementação: Cálculo de Métricas}
\begin{lstlisting}[language=Python, caption=Métricas de validação, basicstyle=\ttfamily\footnotesize]
import numpy as np

def calculate_metrics(y_true, y_pred, n_params):
    """Calcula varias metricas de ajuste"""
    n = len(y_true)

    # Residuos
    residuals = y_true - y_pred

    # SSE e MSE
    sse = np.sum(residuals**2)
    mse = sse / n
    rmse = np.sqrt(mse)

    # R-squared
    ss_tot = np.sum((y_true - np.mean(y_true))**2)
    r2 = 1 - (sse / ss_tot)

    # R-squared ajustado
    r2_adj = 1 - (1-r2)*(n-1)/(n-n_params-1)

    # AIC e BIC
    aic = n*np.log(sse/n) + 2*n_params
    bic = n*np.log(sse/n) + n_params*np.log(n)

    return {
        'SSE': sse, 'MSE': mse, 'RMSE': rmse,
        'R2': r2, 'R2_adj': r2_adj,
        'AIC': aic, 'BIC': bic,
        'residuals': residuals
    }

# Usar
metrics = calculate_metrics(y_data, y_pred, n_params=2)
for key, val in metrics.items():
    if key != 'residuals':
        print(f"{key}: {val:.4f}")
\end{lstlisting}
